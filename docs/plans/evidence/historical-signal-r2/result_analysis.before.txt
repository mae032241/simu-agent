def analysis_parentage(inputs: tuple[Any, ...], parameters: Mapping[str, Any]) -> bool:
    if parameters:
        return False
    grouped: dict[str, list[Any]] = {}
    for item in inputs:
        grouped.setdefault(item.port_name, []).append(item.artifact)
    if any(len(grouped.get(name, ())) != 1 for name in (
        "experiment_plan", "experiment_review", "reviewed_package", "runtime_manifest"
    )):
        return False
    plan, review, package, manifest = (grouped[name][0] for name in (
        "experiment_plan", "experiment_review", "reviewed_package", "runtime_manifest"
    ))
    if not (
        plan.ref in review.parent_refs and review.handoff_verdict == "pass"
        and plan.ref in package.parent_refs and package.ref in manifest.parent_refs
    ):
        return False
    diagnostics = grouped.get("diagnostics", ())
    manifest_labels = dict(manifest.labels)
    if any(
        diagnostic.parent_refs != manifest.parent_refs
        or dict(diagnostic.labels).get("logical_name") != "tcad_log"
        or not manifest_labels.get("execution_id")
        or dict(diagnostic.labels).get("execution_id") != manifest_labels["execution_id"]
        or diagnostic.media_type.split(";", 1)[0] != "text/plain"
        for diagnostic in diagnostics
    ):
        return False
    outputs = grouped.get("solver_outputs", ())
    if any(output.parent_refs != manifest.parent_refs for output in outputs):
        return False
    return all(
        package.ref in audit.parent_refs and manifest.ref in audit.parent_refs
        and all(output.ref in audit.parent_refs for output in outputs)
        for audit in grouped.get("runtime_attestation", ())
    )


def _source_aliases(request: dict[str, Any]) -> tuple[str, ...]:
    mappings = request.get("sources")
    if not isinstance(mappings, (list, tuple)) or not 1 <= len(mappings) <= 40:
        raise ValueError("source_mapping_limit")
    if any(not isinstance(mapping, Mapping) or not isinstance(mapping.get("input_alias"), str) for mapping in mappings):
        raise ValueError("invalid_input_alias")
    names = tuple(dict.fromkeys(mapping["input_alias"] for mapping in mappings))
    if any(not isinstance(name, str) or not name for name in names):
        raise ValueError("invalid_input_alias")
    return names


def parse_tcad_sources(request: dict[str, Any], sources: Mapping[str, bytes], *, deadline: float | None = None) -> CurveBundle:
    """Parse bound bytes without an experiment-time comparison contract."""
    aliases = _source_aliases(request)
    series = []
    for mapping in request["sources"]:
        check_deadline(deadline)
        raw = sources[mapping["input_alias"]]
        if len(raw) > 32 * 1024 * 1024:
            raise ValueError("source_byte_limit")
        admit_raw_lines(raw)
        kind = mapping["format"]
        allowed = {"input_alias", "format", "series_key", "case_key", "role",
                   "x_axis", "y_axis", "output_name", "experiment_key"}
        if kind == "csv":
            allowed.update({"x_column", "y_column"})
        elif kind == "sprocess_plx":
            allowed.update({"dataset_name", "min_points"})
        elif kind == "sprocess_log":
            allowed.add("min_points")
        if set(mapping) - allowed:
            raise NotImplementedError("source_options_unsupported")
        if kind == "csv":
            series.append(csv_series(mapping, raw, deadline=deadline))
        elif kind in {"sprocess_plx", "sprocess_log"}:
            specification = SProcessSeriesSpec.model_validate_json(canonical_json({
                **{key: mapping[key] for key in ("series_key", "case_key", "role", "x_axis", "y_axis")},
                "min_points": mapping.get("min_points", 2), "max_points": MAX_POINTS,
            }), strict=True)
            if kind == "sprocess_plx":
                # The selected physical quantity is explicit, even for a single dataset.
                bundle, _ = normalize_sprocess_plx(raw, SProcessPLXSourceSpec(
                    series=specification, dataset_name=mapping["dataset_name"],
                    max_input_bytes=32 * 1024 * 1024,
                ), source_name=mapping["input_alias"])
            else:
                bundle, _ = normalize_sprocess_log(raw, SProcessLogSourceSpec(
                    expected_series=(specification,), max_input_bytes=32 * 1024 * 1024,
                ))
            series.extend(bundle.series)
        else:
            raise NotImplementedError("source_format_unsupported")
        check_deadline(deadline)
        if sum(len(item.points) for item in series) > MAX_POINTS:
            raise NotImplementedError("source_point_limit")
    return CurveBundle(
        source_profile="tcad.analysis.raw.v1",
        source_digests=tuple(dict.fromkeys(hashlib.sha256(sources[name]).hexdigest() for name in aliases)),
        series=tuple(series),
    )


def evaluate_tcad_request(*, record_key: str, request: dict[str, Any], sources: Mapping[str, bytes], deadline: float | None = None) -> CalculationRecord:
    AnalysisScoreInput(record_key=record_key, request=request)
    digests = {}
    try:
        check_deadline(deadline)
        aliases = _source_aliases(request)
        digests = {name: hashlib.sha256(sources[name]).hexdigest() for name in aliases if name in sources}
        if set(request) != {"sources", "comparison_spec"}:
            raise NotImplementedError("request_options_unsupported")
        bundle = parse_tcad_sources(request, sources, deadline=deadline)
    except TimeoutError:
        return CalculationRecord(record_key=record_key, request=request, input_digests=digests,
            algorithm_version=ALGORITHM_VERSION, status="error", reason_code="calculation_time_budget")
    except (NotImplementedError, ValidationError) as error:
        return CalculationRecord(
            record_key=record_key, request=request, input_digests=digests,
            algorithm_version=ALGORITHM_VERSION, status="unsupported",
            reason_code=str(error) if isinstance(error, NotImplementedError) else "tcad_layout_unsupported",
        )
    except (ValueError, KeyError, TypeError, UnicodeError, csv.Error):
        return CalculationRecord(
            record_key=record_key, request=request, input_digests=digests,
            algorithm_version=ALGORITHM_VERSION, status="unavailable", reason_code="tcad_input_or_mapping_invalid",
        )
    return evaluate_analysis_request(record_key=record_key, request=request, sources=sources, bundle=bundle, deadline=deadline)


def tcad_score_tool(request: AnalysisScoreInput, context: OperationToolContext) -> dict[str, Any]:
    return run_score_tool(request, context, evaluate_tcad_request)


def _identity_context(sources: Mapping[str, bytes]):
    """Read admitted evidence for output comparisons, without re-admitting it."""
    descriptors = getattr(sources, "binding_descriptors", None)
    if descriptors is None or set(descriptors) != set(sources):
        raise RunCheckerError("TCAD analysis requires exact input binding descriptors")
    plan = parse_bound_json(ExperimentPortfolio, sources["experiment_plan"])
    package = parse_bound_json(ReviewedDeckPackage, sources["reviewed_package"])
    manifest = parse_bound_json(TCADRuntimeManifest, sources["runtime_manifest"])
    expected = {output.name: output for output in package.project.expected_outputs}
    known = {alias: expected.get(descriptor.output_name) for alias, descriptor in descriptors.items()
             if descriptor.port_name == "solver_outputs"}
    return plan, package, manifest, descriptors, known


def validate_analysis_inputs(sources: Mapping[str, bytes]) -> None:
    """Deterministic input premises, invoked only before a Run is created."""
    plan, package, manifest, descriptors, _ = _identity_context(sources)
    records = {record.name: record for record in manifest.outputs}
    expected = {output.name: output for output in package.project.expected_outputs}
    cases = {(proposal.experiment_key, case.case_key) for proposal in plan.proposals for case in proposal.cases}
    for output in expected.values():
        if output.case_key is not None and (output.experiment_key, output.case_key) not in cases:
            raise OperationInvocationError("input_package_case_unknown", port="reviewed_package", field="expected_outputs")
    for descriptor in descriptors.values():
        if descriptor.port_name != "solver_outputs":
            continue
        record = records.get(descriptor.output_name)
        if record is None or (record.media_type, record.size_bytes, record.sha256) != (
            descriptor.media_type, descriptor.size_bytes, descriptor.sha256
        ):
            raise OperationInvocationError("input_manifest_output_mismatch", port="solver_outputs")
        declaration = expected.get(record.name)
        if declaration is not None and (declaration.relative_path != record.relative_path
                or declaration.media_type != record.media_type):
            raise OperationInvocationError("input_project_output_mismatch", port="runtime_manifest", field="outputs")


def _check_source_identity(reference, descriptors, known) -> None:
    alias = reference.input_alias if hasattr(reference, "input_alias") else reference["input_alias"]
    values = reference.model_dump() if hasattr(reference, "model_dump") else reference
    if alias not in descriptors:
        raise SemanticRuleViolation("analysis source is not a bound input")
    descriptor = descriptors[alias]
    if descriptor.port_name != "solver_outputs":
        if values.get("output_name") is not None:
            raise SemanticRuleViolation("non-solver evidence cannot claim a registered solver output identity")
        return
    if values.get("output_name") is not None and values["output_name"] != descriptor.output_name:
        raise SemanticRuleViolation("analysis output name differs from exact bound output")
    if values.get("case_key") is not None:
        declaration = known.get(alias)
        if declaration is None or declaration.case_key is None or (
            values.get("experiment_key"), values["case_key"]
        ) != (declaration.experiment_key, declaration.case_key):
            raise SemanticRuleViolation("analysis case differs from explicit project output case")


def analysis_context(payload: dict[str, Any], sources: Mapping[str, bytes], handoff: dict[str, Any]) -> None:
    del handoff
    report = LayeredDiagnosisReport.model_validate_json(canonical_json(payload), strict=True)
    plan, package, manifest, descriptors, known = _identity_context(sources)
    if report.study_kind != plan.study_kind or not any(
        validation.plan_key == report.plan_key and validation.experiment_key == report.experiment_key
        for validation in plan.validation_plans
    ):
        raise SemanticRuleViolation("analysis does not identify the exact bound plan")
    references = {reference.source_key: reference for reference in report.source_references}
    if len(references) != len(report.source_references):
        raise SemanticRuleViolation("analysis source references must be unique")
    evidence_keys = {evidence.source_key for evidence in report.evidence}
    if set(references) - evidence_keys:
        raise SemanticRuleViolation("analysis source reference lacks matching evidence")
    for reference in references.values():
        _check_source_identity(reference, descriptors, known)
    records = {record.record_key: record for record in report.calculation_records}
    for evidence in report.evidence:
        if evidence.source_key in records:
            if evidence.locator != "calculation_records:" + evidence.source_key:
                raise SemanticRuleViolation("calculation evidence requires its canonical record locator")
            if records[evidence.source_key].status != "computed":
                raise SemanticRuleViolation("uncomputed record cannot support an evidence claim")
            continue
        reference = references.get(evidence.source_key)
        if reference is None or evidence.locator != reference.input_alias:
            raise SemanticRuleViolation("TCAD evidence must use an exact alias and structured source reference")
    for record in report.calculation_records:
        mappings = record.request.get("sources", ())
        if not isinstance(mappings, (list, tuple)):
            if record.status == "computed":
                raise SemanticRuleViolation("computed source mappings must be a list")
            mappings = ()
        for mapping in mappings:
            if not isinstance(mapping, Mapping) or not isinstance(mapping.get("input_alias"), str):
                if record.status == "computed":
                    raise SemanticRuleViolation("computed source mapping is invalid")
                continue
            alias = mapping["input_alias"]
            if alias not in descriptors and record.status != "computed":
                continue
            _check_source_identity(mapping, descriptors, known)
            if descriptors[alias].port_name == "solver_outputs" and record.status == "computed":
                if any(mapping.get(key) is None for key in ("output_name", "experiment_key", "case_key")):
                    raise SemanticRuleViolation("computed solver mapping requires explicit output and case identity")
            kind = mapping.get("format")
            media = descriptors[alias].media_type
            if kind == "sprocess_plx" and media != "application/x-synopsys-plx":
                raise SemanticRuleViolation("PLX mapping differs from registered media type")
            if kind == "sprocess_log" and media.split(";", 1)[0] != "text/plain":
                raise SemanticRuleViolation("log mapping differs from registered media type")
            if kind == "csv" and media.split(";", 1)[0] not in {"text/csv", "application/csv"}:
                raise SemanticRuleViolation("CSV mapping differs from registered media type")
        if record.status == "error":
            replay_calculation(record, sources)
        else:
            replay = evaluate_tcad_request(record_key=record.record_key, request=record.request, sources=sources, deadline=getattr(sources, "validation_deadline", None))
            if replay.status == "error":
                raise SemanticRuleViolation("calculation replay budget exhausted; remove this calculation and submit limited analysis")
            if canonical_json(record) != canonical_json(replay):
                raise SemanticRuleViolation("TCAD calculation differs from raw-byte replay")
    validate_analysis_report(report, plan)
    audit_failed = False
    if "runtime_attestation" in sources:
        audit = parse_bound_json(RuntimeAttestation, sources["runtime_attestation"])
        audit_failed = audit.verdict != "pass"
    incomplete = any(output.required and output.name not in {
        descriptor.output_name for descriptor in descriptors.values() if descriptor.port_name == "solver_outputs"
    } for output in package.project.expected_outputs)
    if manifest.terminal_state != "succeeded" or manifest.exit_code != 0 or audit_failed:
        if report.gates.numerical_validity.status == "pass" or report.claim_allowed:
            raise SemanticRuleViolation("failed execution cannot be reported as numerical success")
    unmapped = any(descriptor.port_name == "solver_outputs" and (known.get(alias) is None or known[alias].case_key is None)
                   for alias, descriptor in descriptors.items())
    if (incomplete or unmapped) and (report.claim_allowed or report.overall_verdict == "pass"):
        raise SemanticRuleViolation("missing required outputs prevent a complete successful analysis")

def _ref(name: str, plugin: str | None = None) -> ComponentRef:
    return ComponentRef(name, plugin_id=plugin)


def _input(name: str, schema: str, resource: ComponentRef, *, count: int = 1,
           optional: bool = False, max_bytes: int, inventory: bool = False,
           exposure: str = "full") -> InputPortSpec:
    return InputPortSpec(
        name=name, description=f"Exact bound TCAD analysis {name}.", schema=schema,
        media_types=("*/*",) if inventory else ("application/json",),
        codec=_ref("opaque_codec" if inventory else "json_codec", "general_science"),
        schema_resource=_ref("wildcard_schema", "general_science") if inventory else resource, min_items=0 if optional else 1, max_items=count,
        max_item_bytes=max_bytes, exposure=exposure,
        usage="evidence_inventory" if inventory else "prior_signal",
    )
